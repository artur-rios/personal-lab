{{- /* Overrides the nuno theme's nav to add one level of dropdown: a main menu entry with children (parent = '' in hugo.toml) renders as a button that opens a panel listing them, each with its params.blurb. Everything else is the theme's nav, unchanged. On mobile the dropdown is hidden like the theme's other links, and footer.html lists the children instead. */ -}} {{- $brand := .Site.Params.brand | default (replaceRE "^https?://" "" (trim .Site.BaseURL "/")) -}}
{{ $brand }} {{- range .Site.Menus.main }} {{- if .HasChildren }} {{- $open := false -}} {{- range .Children }} {{- if partial "nav-current.html" (dict "page" $ "entry" .) }}{{ $open = true }}{{ end -}} {{- end }}
{{- range .Children }} {{ .Name }} {{- with .Params.blurb }}{{ . }}{{ end }} {{- end }}
{{- else }} {{- $current := partial "nav-current.html" (dict "page" $ "entry" .) -}} {{- $mobile := eq (lower .Identifier) "about" -}} {{ .Name }} {{- end }} {{- end }} {{- /* Language switcher. Only on a multilingual build, and only listing the languages this page actually exists in — offering a link to a page that is not translated sends the reader to a 404 rather than to a translation. Plain links, not a dropdown: with two or three languages a menu is more machinery than the job needs. */ -}} {{- with .Translations }} {{- range . }} {{ .Language.Label | default .Language.Lang }} {{- end }} {{- end }} {{- /* Desktop: a search pill. Mobile: a round icon. Both open the overlay. */}}
